cmake_minimum_required(VERSION 3.25)
project(win32compat LANGUAGES CXX)
if(NOT TARGET pcmodern::core)
    add_subdirectory(../pcmodern ${CMAKE_CURRENT_BINARY_DIR}/pcmodern)
endif()
add_library(win32compat_kernel32 STATIC src/time.cpp src/fs.cpp)
add_library(win32compat::kernel32 ALIAS win32compat_kernel32)
target_include_directories(win32compat_kernel32 PUBLIC include)
target_link_libraries(win32compat_kernel32 PUBLIC pcmodern::core)
target_compile_features(win32compat_kernel32 PUBLIC cxx_std_17)
